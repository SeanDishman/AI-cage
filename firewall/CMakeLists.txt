cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cage_firewall LANGUAGES CXX)
    include(CTest)
endif()

add_library(cage_firewall src/policy.cpp)
target_include_directories(cage_firewall PUBLIC include)
target_compile_features(cage_firewall PUBLIC cxx_std_20)
if(WIN32)
    target_link_libraries(cage_firewall PUBLIC ws2_32)
else()
    target_sources(cage_firewall PRIVATE src/guard_linux.cpp)
    add_executable(cage-firewall src/main.cpp)
    target_link_libraries(cage-firewall PRIVATE cage_firewall)
endif()
if(MSVC)
    target_compile_options(cage_firewall PRIVATE /W4)
else()
    target_compile_options(cage_firewall PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
    add_executable(firewall_policy_test tests/policy_test.cpp)
    target_link_libraries(firewall_policy_test PRIVATE cage_firewall)
    add_test(NAME firewall_policy COMMAND firewall_policy_test)
endif()
