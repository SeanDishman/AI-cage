cmake_minimum_required(VERSION 3.20)
project(AICage VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(WIN32)
  add_executable(AICage src/windows_launcher.cpp)
  target_compile_definitions(AICage PRIVATE UNICODE _UNICODE)
  if(MINGW)
    target_link_options(AICage PRIVATE -municode -static)
  endif()
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  add_executable(aicage src/controller.cpp)
  add_executable(cage-broker src/broker.cpp)
  target_link_libraries(cage-broker PRIVATE util)
  target_link_options(cage-broker PRIVATE -static)
  add_executable(cage-monitor src/monitor.cpp)
  add_executable(cage-probe tests/probe.cpp)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/firewall/CMakeLists.txt")
    add_subdirectory(firewall)
  endif()
endif()
