cmake_minimum_required(VERSION 3.20)
project(AICageUI VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The local UI launcher targets Windows. The sandbox itself runs in WSL2.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(AICageUI src/server.cpp src/process.cpp)
target_include_directories(AICageUI SYSTEM PRIVATE vendor)
target_compile_definitions(AICageUI PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(AICageUI PRIVATE ws2_32 bcrypt shell32)
if(MINGW)
  target_link_options(AICageUI PRIVATE -static -static-libgcc -static-libstdc++)
  target_compile_options(AICageUI PRIVATE -Wall -Wextra)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(ui-controller-fixture tests/controller_fixture.cpp)
  target_include_directories(ui-controller-fixture SYSTEM PRIVATE vendor)
  target_link_libraries(ui-controller-fixture PRIVATE shell32)
  if(MINGW)
    target_link_options(ui-controller-fixture PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
